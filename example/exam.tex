% Example exam for omr_grader (synthetic: made-up questions, no real course).
%
% The grader reads the key from this file: questions are numbered in document
% order and the answer is the \correctchoice (or \CorrectChoice). Things it
% handles, shown here:
%   - two columns, then a single-column page whose numbering carries on (Q19-20)
%   - code listings: their contents are ignored
%   - a commented-out question (after Q11): not counted
%   - a question with no correct choice (Q15): canceled, with a warning
% Compile with pdflatex (twice) to refresh exam.aux, which the grader uses to
% check the question count.

% the question counter restarts with each questions environment, so carry it
% across the single-column interruptions by hand
\newcounter{savedq}
\newcommand{\qsave}{\setcounter{savedq}{\value{question}}}
\newcommand{\qrestore}{\setcounter{question}{\value{savedq}}}

\renewcommand{\questionshook}{%
  \setlength{\itemsep}{6pt}\setlength{\parsep}{0pt}\setlength{\topsep}{2pt}%
  \interlinepenalty=1000 }
\renewcommand{\choiceshook}{%
  \setlength{\itemsep}{0pt}\setlength{\parsep}{0pt}\setlength{\topsep}{2pt}%
  \interlinepenalty=1000
  % reduce the indent of the options (the class default is 2.5em)
  \settowidth{\leftmargin}{W.\hskip\labelsep\hskip 1em}%
  \labelwidth\leftmargin \advance\labelwidth-\labelsep }

\@beginparpenalty=10000
\@itempenalty=10000
\makeatother

}
\end{lstlisting}

    \choice 3.5
    \correctchoice 3
    \choice 4
    \choice 7

#include <stdio.h>

int main(void)
{
      int sum = 0;
      for (int i = 1; i <= 4; i++)
            sum += i;
      printf("%d\n", sum);
      return 0;
}
\end{lstlisting}

\begin{choices}

    \choice 6

    \choice 15
